api/handler.ts TypeScript 
import { Router } from "vkt";
import { db } from "vkt/storage";

const app = Router();

// Runs at the edge, closest to the user
app.get("/api/users/:id", async (req) => {
  const user = await db.get(`users:${req.params.id}`);

  if (!user) {
    return Response.json(
      { error: "Not found" },
      { status: 404 }
    );
  }

  return Response.json({ user });
});
